"""Recover a current source snapshot; verify it against saved training contracts and test windows."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
OUT = ROOT / 'docs/evidence/pest-recovery-2026-10-22'
YEAR, WEEK = 'Observation Year', 'Standard Week'
WEATHER = ['MaxT(°C)', 'MinT(°C)', 'RH1(%)', 'RH2(%)', 'RF(mm)', 'WS(kmph)', 'SSH(hrs)', 'EVP(mm)']
PESTS = ['Jassid', 'Aphid']
contract = json.loads((ROOT / 'docs/evidence/module-3-candidate/model_contract.json').read_text())
original_provenance = json.loads((ROOT / 'docs/evidence/module-3-candidate/data_provenance.json').read_text())
frames = {}
for pest in PESTS:
    tables = json.loads((OUT / (pest + '_table.json')).read_text())
    assert len(tables) == 1 and len(tables[0]) == 261
    rows = tables[0]
    assert rows[0] == [YEAR, WEEK, 'Pest Value', *WEATHER]
    assert all(len(row) == len(rows[0]) for row in rows)
    frame = pd.DataFrame(rows[1:], columns=rows[0]).apply(pd.to_numeric, errors='coerce')
    assert frame.isna().sum().to_dict() == original_provenance[pest]['missing']
    assert not frame.duplicated([YEAR, WEEK]).any()
    assert set(frame[YEAR]) == {2005, 2006, 2007, 2008, 2009}
    frame.to_csv(OUT / (pest + '_raw.csv'), index=False)
    frames[pest] = frame.rename(columns={'Pest Value': pest})
np.testing.assert_allclose(frames['Jassid'][[YEAR, WEEK, *WEATHER]], frames['Aphid'][[YEAR, WEEK, *WEATHER]])
df = frames['Jassid'].merge(frames['Aphid'][[YEAR, WEEK, 'Aphid']], on=[YEAR, WEEK], validate='one_to_one')
df = df.sort_values([YEAR, WEEK]).reset_index(drop=True)
train_rows = df[YEAR].isin([2005, 2006, 2007])
for pest in PESTS:
    low, high = float(df.loc[train_rows, pest].min()), float(df.loc[train_rows, pest].max())
    assert {'min': low, 'max': high} == contract['pest_scaling'][pest]
    df[pest + '_norm'] = ((df[pest] - low) / (high - low)).clip(0, 1)
df['target_pressure'] = df[[p + '_norm' for p in PESTS]].max(axis=1, skipna=False)
for pest in PESTS:
    df[pest + '_past'] = df.groupby(YEAR)[pest + '_norm'].ffill(limit=4)
df['pest_pressure'] = df[[p + '_past' for p in PESTS]].max(axis=1, skipna=False)
thresholds = np.quantile(df.loc[train_rows, 'target_pressure'].dropna(), [1 / 3, 2 / 3])
np.testing.assert_allclose(thresholds, contract['thresholds'], atol=1e-12, rtol=0)
features = contract['feature_order']
records, test_samples = [], []
for year, season in df.groupby(YEAR, sort=True):
    season = season.sort_values(WEEK)
    for i in range(4, len(season)):
        segment = season.iloc[i - 4:i + 1]
        if not np.all(np.diff(segment[WEEK]) == 1):
            continue
        past = segment.iloc[:-1][features].to_numpy(dtype='float32')
        target = float(segment.iloc[-1]['target_pressure'])
        if not np.isfinite(past).all() or not np.isfinite(target):
            continue
        label = int(np.digitize(target, thresholds))
        sample_id = f'{int(year)}-W{int(segment.iloc[-1][WEEK]):02d}'
        split = 'train' if year < 2008 else 'validation' if year == 2008 else 'test'
        records.append({'sample_id': sample_id, 'split': split, 'class_index': label})
        if split == 'test':
            test_samples.append({'label': sample_id, 'sequence': past.tolist(), 'actual_class': contract['class_order'][label]})
saved = json.loads((ROOT / 'models/sample_sequences.json').read_text())['samples']
assert [s['label'] for s in saved] == [s['label'] for s in test_samples]
assert [s['actual_class'] for s in saved] == [s['actual_class'] for s in test_samples]
np.testing.assert_allclose([s['sequence'] for s in saved], [s['sequence'] for s in test_samples], atol=1e-6, rtol=0)
df.to_csv(OUT / 'processed_weekly_data.csv', index=False)
splits = pd.DataFrame(records)
splits.to_csv(OUT / 'reconstructed_split_manifest.csv', index=False)
report = {
    'retrieved_utc': datetime.now(timezone.utc).isoformat(),
    'source': 'Current public ICAR-CRIDA CropPest DSS pages, extracted with gstack browser; archived DOM is not the original HTTP byte stream.',
    'queries': {p: original_provenance[p]['query'] for p in PESTS},
    'verification': {'raw_counts_and_missingness_match_original_record': True,
                     'training_pest_scaling_matches_contract': True,
                     'training_thresholds_match_contract': True,
                     'all_19_exported_test_sequences_and_labels_match': True},
    'split_counts': splits.groupby('split').size().to_dict(),
    'limitations': 'A newly retrieved and verified reconstruction, not a recovered original run export or a new blind evaluation. Source pages use HTTP and their current integrity is not independently certified.',
    'sha256': {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(OUT.iterdir()) if p.is_file()},
}
(OUT / 'recovery_verification.json').write_text(json.dumps(report, indent=2, allow_nan=False))
print(json.dumps(report, indent=2))
